#include <platform.h>

#include <uart.h>
#include <timer.h>
#include <string.h>
#include "delay.h"
#include "lcd.h"

static volatile int changed;
static volatile unsigned int secs;
static volatile unsigned int centiseconds;  // 1/100s of a second
static volatile unsigned int minutes;

void systick_isr(void){
    secs++;
    changed = 1;
}
void new_systick_isr(void) {
        centiseconds++;  // Increment centiseconds (10ms units)
        
        // Handle rollover from centiseconds to seconds
        if (centiseconds >= 100) {
            centiseconds = 0;  // Reset centiseconds
            secs++;  // Increment seconds
            
            // Handle rollover from seconds to minutes
            if (secs >= 60) {
                secs = 0;  // Reset seconds
                minutes++;  // Increment minutes
            }
        }
    
    // Set flag to indicate time has been updated (triggers display refresh)
    changed = 1;
}
void main_sample(void) {
    char s[20];
    // initialise LCD
    lcd_init();	
	lcd_clear();
    lcd_print("EBU5477 Lab 3");

    // set up systick timer
    timer_init(1000000);
    timer_set_callback(systick_isr); // check the ISR
    timer_enable();
    changed = 0; // flag for the display
    secs = 0;
    
    __enable_irq();
    
    while(1) {  
        if (changed){
            changed = 0;
            sprintf(s, "%d", secs);
            lcd_set_cursor(0, 1);
            lcd_print(s); 
        }
    }
}
void main1(void) {
    char s[20];
    
    // Initialize LCD display
    lcd_init();
    lcd_clear();
    lcd_print("EBU5477 Lab 3");  // Display lab title
    
    // Configure SysTick timer for 100Hz (10ms intervals)
    // Timer period = 1000000 / 100 = 10000 ticks (10ms per interrupt)
    timer_init(10000);  
    timer_set_callback(new_systick_isr);  // Use timer interrupt for time updates
    timer_enable();
    
    // Reset display update flag and time counters
    changed = 0;  
    centiseconds = 0;
    secs = 0;
    minutes = 0;

    __enable_irq();  // Enable global interrupts
    
    while(1) {  
        // Update display when timer interrupt flags a change
        if (changed) {
            changed = 0;  // Reset flag after processing
            // Format time as HH:MM:SS.ss (hours:minutes:seconds.centiseconds)
            sprintf(s, "%02d:%02d.%02d", minutes, secs, centiseconds);
            lcd_set_cursor(0, 1);  // Position cursor on second line
            lcd_print(s);  // Display formatted time
        }
    }
}
void main2(void) {
    char s[20];
    
    // Initialize LCD and UART for input/output
    lcd_init();
    lcd_clear();
    uart_init(9600);  // Set UART baud rate
    uart_enable();
    
    // Target name to be typed (case-sensitive)
    const char* target_name = "zuozecheng";
    int target_len = strlen(target_name);  // Length of target name
    int match_idx = 0;  // Current character index in input
    int timer_running = 0;  // Flag to control timer
    char input[20] = {0};  // Buffer to store user input
    
    // Configure timer for 100Hz (10ms intervals) for timing
    timer_init(10000);
    timer_set_callback(new_systick_isr);
    timer_enable();
    
    // Reset time counters and display flag
    changed = 0;
    centiseconds = 0;
    secs = 0;
    minutes = 0;

    __enable_irq();  // Enable interrupts
    
    // Initial display prompt
    lcd_clear();
    lcd_set_cursor(0, 0);
    lcd_print("Enter my name:");  // Prompt user to type the name
    lcd_set_cursor(0, 1);  // Position cursor on second line for input display
    
    while (1) {
        // Read character from UART (blocking if no input)
        char c = uart_rx();  
        
        // Process valid lowercase letters
        if (c >= 'a' && c <= 'z') {
            // Start timer on first character input
            if (!timer_running) {
                timer_running = 1;
                // Reset time and input state
                centiseconds = 0;
                secs = 0;
                minutes = 0;
                match_idx = 0;
                memset(input, 0, sizeof(input));  // Clear input buffer
                lcd_clear();  // Clear display for new attempt
                lcd_set_cursor(0, 0);  // Reposition to prompt
            }
            
            // Append character to input and display it
            input[match_idx++] = c;
            lcd_set_cursor(match_idx - 1, 0);  // Update cursor position
            char temp[2] = {c, '\0'};  // Convert to C-string
            lcd_print(temp);  // Display the typed character
            
            // Check if input matches target name
            if (match_idx == target_len) {
                input[match_idx] = '\0';  // Null-terminate input string
                
                if (strcmp(input, target_name) == 0) {  // Match success
                    timer_running = 0;  // Stop timer
                    lcd_clear();
                    lcd_set_cursor(0, 0);
                    lcd_print("Success!");  // Display success message
                    // Format and display total time taken
                    sprintf(s, "Time: %02d:%02d.%02d", minutes, secs, centiseconds);
                    lcd_set_cursor(0, 1);
                    lcd_print(s);
                    delay_ms(3000);  // Hold result for 3 seconds
                } else {  // Match failure
                    timer_running = 0;  // Stop timer
                    lcd_clear();
                    lcd_set_cursor(0, 0);
                    lcd_print("Incorrect, try again");  // Display error message
                    delay_ms(1000);  // Hold message for 1 second
                }
                
                // Reset for new attempt
                lcd_clear();
                lcd_set_cursor(0, 0);
                lcd_print("Enter my name:");  // Redisplay prompt
                lcd_set_cursor(0, 1);
                match_idx = 0;
                memset(input, 0, sizeof(input));
            }
        }
        
        // Update timer display while typing is in progress
        if (changed && timer_running) {
            changed = 0;  // Reset flag
            // Format and display elapsed time
            sprintf(s, "Timer: %02d:%02d.%02d", minutes, secs, centiseconds);
            lcd_set_cursor(0, 1);  // Update timer on second line
            lcd_print(s);
        }
    }
}

void main3(void) {
    // Initialize LCD and configure button GPIO (PA_0 as pull-up input)
    lcd_init();
    lcd_clear();
    gpio_set_mode(PA_0, PullUp);  // Button connected to PA_0 (press pulls to GND)
    
    // Configure timer for 100Hz (10ms intervals) for reaction time measurement
    timer_init(10000);
    timer_set_callback(new_systick_isr);
    timer_enable();
    
    // Reset global time counters and display flag
    centiseconds = 0;
    secs = 0;
    minutes = 0;
    changed = 0;

    __enable_irq();  // Enable interrupts
    
    char s[20];  // Buffer for formatted output
    
    while (1) {
        // Initial state: Prompt user to press button to start
        lcd_clear();
        lcd_print("Press button");
        lcd_set_cursor(0, 1);
        lcd_print("to begin");
        
        // Wait for button press (active-low due to pull-up)
        while (gpio_get(PA_0) == 1);  // Wait for button down
        delay_ms(200);  // Debounce delay
        while (gpio_get(PA_0) == 0);  // Wait for button release
        
        // Display "Get ready..." for 1 second
        lcd_clear();
        lcd_print("Get ready...");
        delay_ms(1000);  // Fixed 1-second countdown
        
        // Generate random delay between 1-3 seconds to prevent anticipation
        int seed = centiseconds + secs * 100;  // Use current time as seed
        // Linear congruential generator for pseudo-random number
        int delay_time = 1000 + ((seed * 1103515245) % 2000);  // 1000-3000ms delay
        delay_ms(delay_time);
        
        // Display reaction prompt and start timer
        lcd_clear();
        lcd_print("PRESS NOW!");
        // Reset time counters and start timer
        centiseconds = 0;
        secs = 0;
        minutes = 0;
        int timer_running = 1;  // Local flag to control timing
        
        // Wait for button press (reaction time measurement)
        while (gpio_get(PA_0) == 1);  // Wait for button down during reaction phase
        // Calculate reaction time in milliseconds (centiseconds = 10ms units)
        int reaction_time = secs * 100 + centiseconds;  // Total 10ms units × 10ms = ms
        
        // Stop timer and debounce
        timer_running = 0;
        delay_ms(200);  // Debounce delay
        
        // Display reaction time result
        lcd_clear();
        sprintf(s, "Reaction: %d ms", reaction_time * 10);  // Convert to milliseconds
        lcd_print(s);
        lcd_set_cursor(0, 1);
        lcd_print("Press to retry");  // Prompt for next attempt
        
        // Wait for button release and next press to retry
        while (gpio_get(PA_0) == 0);  // Wait for button release
        while (gpio_get(PA_0) == 1);  // Wait for next button press
        delay_ms(200);  // Debounce delay
    }
}
